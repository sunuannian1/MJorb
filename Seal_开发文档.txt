不能使用字符串直接比较：

必须使用版本组件比较。
func isVersion(
    _ current: String,
    atLeast minimum: String
) -> Bool {
    let currentVersion = current
        .split(separator: ".")
        .compactMap { Int($0) }

    let minimumVersion = minimum
        .split(separator: ".")
        .compactMap { Int($0) }

    for index in 0..<max(
        currentVersion.count,
        minimumVersion.count
    ) {
        let currentValue =
            index < currentVersion.count
            ? currentVersion[index]
            : 0

        let minimumValue =
            index < minimumVersion.count
            ? minimumVersion[index]
            : 0

        if currentValue != minimumValue {
            return currentValue > minimumValue
        }
    }

    return true
}

21.2 主流程
func start(
    ipaID: UUID,
    installAfterSigning: Bool
) async throws -> SigningTask {

    let task = try createTask(ipaID: ipaID)

    try updateState(.preparing)

    let ipa = try appStore.loadIPA(id: ipaID)

    try updateState(.extracting)

    let context = try await materialProvider
        .prepareSigningContext(
            for: ipa.bundleIdentifier
        )

    let signedURL = try await signingProvider.sign(
        appURL: ipa.appURL,
        context: context
    )

    try updateState(.verifying)

    try await verifier.verify(
        appURL: signedURL
    )

    try updateState(.packaging)

    let packagedURL = try package(signedURL)

    if installAfterSigning {
        try updateState(.installing)

        try await installationService.install(
            ipaURL: packagedURL
        )
    }

    try updateState(.completed)

    return try finishTask(
        signedURL: packagedURL
    )
}
实际项目中必须根据真实后端调整，不得直接复制后假设可以运行。

四十九、最终验收标准
49.1 启动
	•	App 可以正常启动；
	•	本地数据可以恢复；
	•	不显示假数据；
	•	不出现空白页面；
	•	不出现重复任务。
49.2 导入
	•	可以选择 IPA；
	•	可以解析 Info.plist；
	•	可以识别 Bundle ID；
	•	可以识别版本；
	•	可以识别最低系统版本；
	•	非法 IPA 有明确错误。
49.3 签名
	•	使用真实签名后端；
	•	使用真实证书和私钥；
	•	使用真实 Profile；
	•	正确处理 Entitlements；
	•	正确处理 Framework；
	•	正确处理 Extension；
	•	主 App 最后签名；
	•	签名后进行验证。
49.4 安装
	•	使用真实安装后端；
	•	不把签名成功当作安装成功；
	•	保存真实错误；
	•	安装成功后更新状态；
	•	安装失败后保留诊断信息。
49.5 续签
	•	使用实际有效期；
	•	不硬编码 7 天；
	•	重新获取签名材料；
	•	重新签名；
	•	重新安装；
	•	更新有效期；
	•	失败时保留原记录。
49.6 数据一致性
所有增删改查必须同步更新：
首页
应用页
历史页
应用详情页
统计数量
续签提醒

五十一、给开发 AI 的最终要求
不要只修改 UI。
不要只增加几个按钮。
不要创建假的：
signIPA()
installIPA()
不要把：
签名成功
直接当成：
安装成功
不要把：
7 天
写死成所有应用的有效期。
不要把所有 Entitlements 删除。
不要只签主 App。
不要忽略 Framework 和 Extension。
不要让首页、应用页、历史页各自维护独立数据。
不要在没有真实源码依据的情况下声称“已经修复安装失败”。
必须先检查现有代码，找到：
签名入口
签名引擎
签名材料获取
Profile 处理
Entitlements 处理
IPA 打包
安装入口
设备配对
Lockdown / 安装服务
安装错误映射
然后逐文件改造。
最终验收标准：
用户打开 Seal
    ↓
导入 IPA
    ↓
读取应用信息
    ↓
检查兼容性
    ↓
真实签名
    ↓
真实验证
    ↓
真实安装
    ↓
真实记录
    ↓
真实续签
    ↓
所有页面同步更新
只有这条链路全部走通，才算 Seal 完成。
这份文档已经覆盖了产品流程、数据流、状态机、接口、存储、错误处理和验收。但要继续做到你说的“精确到现有项目每一个文件、每一个函数”，下一步必须对照 Seal 当前源码，尤其是 签名入口、安装入口、Lockdown/配对代码和失败日志。否则继续写具体函数名，只会再次变成看起来完整、实际接不上的空架构。
